# The guests: one line each, of four parts with a tab between.
#
#   name       the program, built from <where>/<name>.c, .m or .swift into <where>/bin (for iOS, <where>/bin/ios)
#   where      examples for the programs that show what ipswrun runs, tests/kernel for those that hold the
#              kernel it stands in for against a Mac's
#   platforms  macos, ios or both: what it is built for and run on
#   how        the flags ipswrun runs it with, and after -- the guest's own arguments; $bin is where the
#              guests of its folder are built and $out where the run's results go. The one word beside
#              is for what is built to be run beside another, and never by itself.
#
# A guest with two lines is run differently on each platform. What it prints is recorded in
# <where>/expected/<name>/<build>.json (for iOS, <where>/expected/ios/<name>). tests/matrix.sh and
# the Makefiles read this file, and nothing else says which guests there are.

# Programs that show what ipswrun runs.
# adi asks the release's adid, which on iOS is another daemon, of FairPlay's, that needs more of an iPhone than there is.
adi	examples	macos	--daemon com.apple.adid=system:/System/Library/PrivateFrameworks/CoreADI.framework/Versions/A/adid
# client is run beside the server whose services it uses.
client	examples	macos,ios	--agent com.example.ipswrun.echo,com.example.ipswrun.reverse=$bin/server
crypto	examples	macos,ios
foundation	examples	macos,ios
guest	examples	macos,ios
# objcinfo describes the frameworks it is given: one that only exists as a private framework.
objcinfo	examples	macos,ios	-- /System/Library/PrivateFrameworks/IMCore.framework/IMCore
# prefs is run beside the release's own cfprefsd, in both of its roles. Two variables of ipswrun's keep CoreFoundation from asking for cfprefsd, and go. On macOS cfprefsd is started when it is first looked up, as its launchd plists say; on iOS, where the agent has no plist of its own, both are started by hand.
prefs	examples	macos	--env __CFPREFERENCES_AVOID_DAEMON --env CFFIXED_USER_HOME --on-demand 'com.apple.cfprefsd.*'
prefs	examples	ios	--env __CFPREFERENCES_AVOID_DAEMON --env CFFIXED_USER_HOME --agent 'com.apple.cfprefsd.agent=system:/usr/sbin/cfprefsd agent' --daemon 'com.apple.cfprefsd.daemon=system:/usr/sbin/cfprefsd daemon'
# server gives client its services.
server	examples	macos,ios	beside
# service is run as the process that provides the service it then uses.
service	examples	macos,ios	--service com.example.ipswrun.echo
# shell runs /bin/sh and /bin/echo, which iOS has not.
shell	examples	macos
swiftasync	examples	macos,ios
swiftrt	examples	macos,ios
